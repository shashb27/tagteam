import { createOpencode } from '@opencode-ai/sdk';
const { client } = await createOpencode();
const session = await client.session.create({ body: { title: 'spike' } });

## 3. `server/agent/opencodeRunner.js` (tasks 0.9, 0.10, 0.14)

```js
import { createOpencodeClient } from '@opencode-ai/sdk';
import path from 'node:path';
import { DEMO_WORKSPACE_DIR, OPENCODE_MODEL, OPENCODE_PROVIDER } from '../config.js';

const ALLOWED_TOOLS = ['Read', 'Grep', 'Glob'];

class OpencodeAgentSession {
  constructor({ sessionId }) {
    this.client = createOpencodeClient({ baseUrl: OPENCODE_BASE_URL });
    this.opencodeSessionId = null;
    this.subscribed = null;
    this.disposed = false;
  }
  async run({ userText, systemPrompt, onEvent, signal }) {
    if (!this.opencodeSessionId) {
      const s = await this.client.session.create({ body: { title: 'tagteam' } });
      this.opencodeSessionId = s.id;
      this.events = await this.client.event.subscribe();
      // pump events in the background → onEvent
      this._pumpEvents(onEvent, signal);
    }
    await this.client.session.prompt({
      path: { id: this.opencodeSessionId },
      body: {
        system: systemPrompt,
        parts: [{ type: 'text', text: userText }],
        tools: ALLOWED_TOOLS,          // opencode enforce allowlist
        // permissions: read-only + cwd confined via opencode.json rules
      },
    });
    // the event pump has already emitted assistant_delta/tool_activity;
    // final text comes from the last message part
    const msgs = await this.client.session.messages({ path: { id: this.opencodeSessionId } });
    const last = msgs.at(-1);
    const text = last.parts.filter(p => p.type === 'text').map(p => p.text).join('');
    return { text, stopReason: 'end_turn' };
  }
  _pumpEvents(onEvent, signal) { /* map opencode events → onEvent; see spike */ }
  dispose() { /* client.session.delete if needed */ }
}
```

**Event mapping (confirmed by spike):** the `_pumpEvents` loop translates opencode SSE into TagTeam's `onEvent` calls:
- `text` part delta → `onEvent({type:'text_delta', text})`
- tool call start → `onEvent({type:'tool_start', tool, summary})`
- tool result → `onEvent({type:'tool_end', tool, summary, ok})`

The rest of `turns.js` is unchanged — it already consumes these `onEvent` shapes.

## 4. `server/agent/index.js` (task 0.10)

```js
import { MOCK_CLAUDE, OPENCODE_BASE_URL } from '../config.js';
import { mockBackend } from './mockRunner.js';

export async function createAgentBackend() {
  if (MOCK_CLAUDE) return mockBackend();
  try {
    const { opencodeBackend } = await import('./opencodeRunner.js');
    return opencodeBackend({ baseUrl: OPENCODE_BASE_URL });
  } catch (err) {
    console.warn('[agent] opencode unavailable:', err.message, '— using mock');
    return mockBackend();
  }
}
```
